cmake_minimum_required(VERSION 3.10)
project(DebayerCPU LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED True)
find_package(Threads REQUIRED)

add_library(debayer_cpu_lib cpu_debayer.cpp cpu_kernel.cpp threadpool.cpp)
target_include_directories(debayer_cpu_lib PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(debayer_cpu_lib PUBLIC Threads::Threads)

option(DEBAYER_CPU_BUILD_DEMO "Build the OpenCV image-directory demo" ON)
if(DEBAYER_CPU_BUILD_DEMO)
    find_package(OpenCV REQUIRED)
    add_executable(debayer_cpu main.cpp)
    target_include_directories(debayer_cpu PRIVATE ${OpenCV_INCLUDE_DIRS})
    target_link_libraries(debayer_cpu PRIVATE debayer_cpu_lib ${OpenCV_LIBS})
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(cpu_regression tests/cpu_regression.cpp)
    target_link_libraries(cpu_regression PRIVATE debayer_cpu_lib)
    add_test(NAME cpu_regression COMMAND cpu_regression)
    set_tests_properties(cpu_regression PROPERTIES TIMEOUT 60)
    add_executable(cpu_menon_reference tests/menon_reference.cpp)
    target_link_libraries(cpu_menon_reference PRIVATE debayer_cpu_lib)
    add_test(NAME cpu_menon_reference COMMAND cpu_menon_reference)
    set_tests_properties(cpu_menon_reference PROPERTIES TIMEOUT 180)
    add_executable(cpu_softmenon_reference tests/softmenon_reference.cpp)
    target_link_libraries(cpu_softmenon_reference PRIVATE debayer_cpu_lib)
    add_test(NAME cpu_softmenon_reference COMMAND cpu_softmenon_reference)
    set_tests_properties(cpu_softmenon_reference PROPERTIES TIMEOUT 180)
    if(DEBAYER_CPU_BUILD_DEMO)
        add_executable(cpu_psnr_regression tests/psnr_regression.cpp)
        target_include_directories(cpu_psnr_regression PRIVATE ${OpenCV_INCLUDE_DIRS})
        target_link_libraries(cpu_psnr_regression PRIVATE debayer_cpu_lib ${OpenCV_LIBS})
        add_test(NAME cpu_psnr_regression COMMAND cpu_psnr_regression)
    endif()
endif()
